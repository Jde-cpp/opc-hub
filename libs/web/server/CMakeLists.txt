cmake_minimum_required( VERSION ${MIN_REQ_CMAKE_VERSION} )
include( ${CMAKE_CURRENT_LIST_DIR}/../../../build/functions.cmake )
project( Jde.Web.Server VERSION ${JDE_VERSION} LANGUAGES CXX )
include( ${jdeRoot}/build/dependencies.cmake )

set( targetName Jde.Web.Server )
add_library( ${targetName} )
compileOptions( ${targetName} )

file( GLOB sources CONFIGURE_DEPENDS *.cpp auth/*.cpp )
file( GLOB headers CONFIGURE_DEPENDS   *.h ${jdeRoot}/include/jde/web/server/*.h )
target_sources( ${targetName} PRIVATE ${sources} ${headers} )

target_link_libraries( ${targetName} protobuf::libprotobuf ) #the imported target on every platform:  the one shared protobuf, which Jde links too.
target_link_libraries( ${targetName} Jde.QL Jde.Web.Client Jde.Access )
add_dependencies( ${targetName} Jde.App.Shared ) #IApp.h pulls in log/ProtoLog.h -> jde/app/proto/Log.pb.h, generated by the protoc step of Jde.App.Shared.

target_precompile_headers( ${targetName}
  PRIVATE
	"pc.h"
)
