cmake_minimum_required( VERSION ${MIN_REQ_CMAKE_VERSION} )
include( ${CMAKE_CURRENT_LIST_DIR}/../../../build/functions.cmake )
project( Jde.App.Shared VERSION ${JDE_VERSION} LANGUAGES CXX )
include( ${jdeRoot}/build/dependencies.cmake )

set( targetName Jde.App.Shared )
add_library( ${targetName} )
compileOptions( ${targetName} )
target_link_libraries( ${targetName} Boost::json fmt::fmt )
add_dependencies( ${targetName} Jde.Web.Client ) #the PCH below includes Web.FromServer.pb.h, generated by Jde.Web.Client's protoc step.

file(GLOB sources CONFIGURE_DEPENDS *.cpp proto/*.cpp log/*.cpp )
file(GLOB headers CONFIGURE_DEPENDS  */*.h ${jdeRoot}/include/jde/app/shared/*.h )
file(GLOB protos CONFIGURE_DEPENDS proto/*.proto )
target_sources( ${targetName} PRIVATE ${sources} ${headers} ${protos} )

if( WIN32 )
	find_package( OpenSSL REQUIRED )
	target_link_libraries( ${targetName} OpenSSL::SSL )
	target_link_libraries( ${targetName} absl::abseil_dll )#every absl:: target aliases it; this library links neither Jde nor protobuf::libprotobuf, which would bring it.
	target_link_libraries( ${targetName} ${CMAKE_INSTALL_PREFIX}/protobuf/lib/protobuf$<IF:$<CONFIG:Debug>,d,>.lib )
endif()

#######################################################
target_precompile_headers(${targetName}
  PUBLIC
	<jde/web/client/proto/Web.FromServer.pb.h>
	<jde/fwk.h>
	PRIVATE
	pc.h
)

#Generated out-of-source, under the build tree's include root (see jdeGeneratedIncludeDir in build/functions.cmake): a
#.pb.h in the source tree is both a protoc output and a header other targets #include, so a consumer's dep-scan can hold
#it open while protoc rewrites it - fatal on windows ("Invalid argument"), latent on linux.  The output dir mirrors the
#include layout, so `<jde/app/proto/X.pb.h>` resolves through the exported root - no source-tree symlinks.
set( protoOutDir ${jdeGeneratedIncludeDir}/jde/app/proto )
file( MAKE_DIRECTORY ${protoOutDir} )
protobuf_generate(
	TARGET ${targetName}
	IMPORT_DIRS proto ../../web/client/proto
	PROTOC_OUT_DIR ${protoOutDir}
)
suppressProtoWarnings( "${protos}" ${protoOutDir} )
#The root for `<jde/app/proto/X.pb.h>`; ${protoOutDir} because the hand-written proto/*.cpp and protoc's own siblings include
#by bare name ("Log.pb.h"); the web/client proto dir because the generated App.*.pb.h import Web.FromServer.proto and so
#include its header by bare name too.  All PUBLIC: consumers include the generated headers, and the PCH above is PUBLIC.
target_include_directories( ${targetName} PUBLIC ${jdeGeneratedIncludeDir} ${protoOutDir} ${jdeGeneratedIncludeDir}/jde/web/client/proto )
